# exedee.me — direct-share landing page, not for search.
#
# WHY THIS FILE ALLOWS CRAWLING WHILE THE PAGE SAYS noindex
# --------------------------------------------------------
# Keeping pages out of search is the job of the <meta name="robots"
# content="noindex"> tag in index.html, NOT of a Disallow rule here.
# A "Disallow: /" would stop crawlers fetching the page, so they would
# never SEE the noindex tag — and the bare URL could still be indexed
# from any inbound link, showing in results with no title or snippet.
# Allowing the crawl is what makes the noindex actually work.
#
# Do not add "Disallow: /" to this file.

User-agent: *
Allow: /

# --- Social preview scrapers -------------------------------------------
# These generate the link card when the URL is shared. They must be able
# to fetch the page or the share card renders blank — which would defeat
# the point of a link posted on LinkedIn.

User-agent: LinkedInBot
Allow: /

User-agent: Twitterbot
Allow: /

User-agent: facebookexternalhit
Allow: /

User-agent: Slackbot-LinkExpanding
Allow: /

User-agent: WhatsApp
Allow: /

# --- AI crawlers --------------------------------------------------------
# Blocked: this page is not meant to be public-facing or citable, so there
# is no reason for it to be absorbed into training or answer engines.
# If exedee.me later becomes a public marketing site, change these four to
# Allow and remove the noindex tag from index.html at the same time.

User-agent: GPTBot
Disallow: /

User-agent: ClaudeBot
Disallow: /

User-agent: PerplexityBot
Disallow: /

User-agent: Google-Extended
Disallow: /

User-agent: CCBot
Disallow: /

# No sitemap is published: a sitemap asks search engines to index the very
# pages this site is telling them to leave alone.
